Overflow: infinite

The scrollbar becomes an ink ball in the margin that prints what you have read. Past the last line, the page keeps going and comes back around.

Scroll inside the page, or drag the ball. With the ball focused, arrow keys draw and Escape sends it home.

// the mark, seen from above
export function glyphOnBall(theta, dir) {
  const c = Math.cos(theta), s = Math.sin(theta);
  return { ox: dir.x * s * R * .78, oy: dir.y * s * R * .78,
           squash: Math.max(.05, c), visible: c > 0 };
}

export const C = TAU * R;           // one revolution of travel
// the mark touches the paper at θ = π
export const firstStamp = theta0 => R * mod(Math.PI - theta0, TAU);
// dash [C − GAP, GAP], each gap centred on a stamp
export const dashOffset = theta0 =>
  mod(C - GAP / 2 - firstStamp(theta0), C);
