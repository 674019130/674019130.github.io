function frame(now) {
  const t = Math.min(1, (now - started) / 420);
  const eased = t * t * (3 - 2 * t);        // smoothstep
  visibleFace = source.map((v, i) => v + (target[i] - v) * eased);
  drawFace(visibleFace);
  if (t < 1) faceFrame = requestAnimationFrame(frame);
}

function stop() { sequenceToken++; clearTimeout(sequenceTimer); clearTimeout(reactionTimer); }
function sequence() {
  stop();
  const token = sequenceToken;
  function next() {
    if (token !== sequenceToken) return;   // a newer action won
    mood(steps[index++]);
    if (index < steps.length) sequenceTimer = setTimeout(next, 1900);
  }
  next();
}
