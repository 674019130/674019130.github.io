function flip(mutate) {
  const before = new Map();
  for (const el of cardEls.values())
    before.set(el, el.getBoundingClientRect()); // visible, mid-flight
  for (const el of before.keys()) flips.get(el)?.cancel();
  mutate();
  for (const [el, a] of before) {
    const b = el.getBoundingClientRect();
    flips.set(el, el.animate([
      { transform: `translate(${a.left - b.left}px, ${a.top - b.top}px)` },
      { transform: 'none' },
    ], { duration: 340, easing }));
  }
}

export function wouldExceed(cards, id, status) {
  const card = cards.find(item => item.id === id);
  if (!card || card.status === status) return false;
  const { count, limit } = load(cards, status);
  return limit != null && count + 1 > limit;
}
