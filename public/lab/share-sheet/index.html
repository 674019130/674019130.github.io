.copy-window {
  display: grid;
  height: 18px;
  overflow: hidden;
}
.copy-face { grid-area: 1 / 1; }

.copy-enter {
  animation: enter 340ms
    cubic-bezier(.2, .85, .25, 1.12) both;
}
.copy-leave {
  animation: leave 340ms
    cubic-bezier(.22, 1, .36, 1) both;
}

@keyframes enter {
  from { transform: translateY(110%); opacity: 0; }
  to   { transform: translateY(0);    opacity: 1; }
}
@keyframes leave {
  from { transform: translateY(0);     opacity: 1; }
  to   { transform: translateY(-110%); opacity: 0; }
}

destination.style.visibility = 'hidden';              // the real chip avatar waits
const dx = to.left - from.left, dy = to.top - from.top, scale = to.width / from.width;
const animation = flight.animate([
  { transform: 'translate(0,0) scale(1)', offset: 0 },
  { transform: `translate(${dx * .6}px,${dy * .68 - 14}px) scale(${1 + (scale - 1) * .55})`, offset: .55 },
  { transform: `translate(${dx}px,${dy}px) scale(${scale})`, offset: 1 },
], { duration: 520, easing: 'cubic-bezier(.25,.85,.25,1)', fill: 'forwards' });
animation.finished.then(() => { destination.style.visibility = ''; flight.remove(); });
