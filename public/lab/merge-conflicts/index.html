Merge conflicts

export function scrambleFrame(text, progress, random = Math.random) {
  const n = Math.floor(progress * progress * text.length); // fast, then slow
  let tail = '';
  for (let i = n; i < text.length; i++)
    tail += text[i] === ' ' ? ' ' : NOISE[Math.floor(random() * NOISE.length)];
  return { head: text.slice(0, n), tail };      // tail is drawn fainter
}
